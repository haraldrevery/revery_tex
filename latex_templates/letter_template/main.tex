%  in the browser with no local TeX Live or MiKTeX install — and the
%  same file still builds on a desktop TeX Live, unchanged.

%                            bibliography or index in a letter)
%
%  THE CONSTRAINTS — why some obvious choices are missing here:
%    • XeLaTeX only        — no LuaLaTeX format ships in the browser engine

%    • No EPS / PS artwork — that needs Ghostscript; use PNG, JPG or PDF.

%    • Text font size   → \RootFontSize below        (10pt | 11pt | 12pt)

%    • Letter spacing   → LetterSpace inside each font block
%    • Accent colour    → \definecolor{AccentColor} in COLOURS

% ============================================================

% WHY THESE FONTS AND NOT Times New Roman / Arial / Garamond
%  Those faces are proprietary fonts installed by your operating
%  system.  The browser engine has no system font database at all —
%  it can only see fonts that ship with the TeX distribution itself.
%  Naming a system font there does not fall back to something
%  similar; it stops the build.
%
%  TeX Gyre is the way out.  Pagella is a *metric clone* of Palatino:
%  warm, wide and elegant — widely considered the most beautiful of
%  the TeX Gyre text faces, and a classic choice for letters.  SIL
%  Open Font Licence; ships with TeX Live, so this file also builds
%  unchanged on your desktop.

  % ---- Text letter spacing ------------------------------------

  LetterSpace = 0,
]

%  Emptiness test: \IfNonEmpty{<field>}{<code>} typesets <code> only
%  when <field> expands to something non-empty.  This is what lets any
%  metadata field in this template be left empty {} without breaking
%  the compile (a bare \\ after an empty line is a LaTeX error).
\makeatletter
\newcommand{\IfNonEmpty}[2]{%
  \def\hrv@test{#1}%
  \if\relax\hrv@test\relax
    \expandafter\@gobble
  \else
    \expandafter\@firstofone
  \fi
  {#2}}
\makeatother

%  Multi-line address blocks: \\ works inside a tabular row.
%  An empty address prints nothing at all.
\newcommand{\AddressBlock}[1]{%
  \IfNonEmpty{#1}{\begin{tabular}{@{}l@{}}#1\end{tabular}}}

%  The letterhead.  Logo space is reserved on the right even when no
%  image exists, so the layout never jumps when you add one.
%  Built from \par-separated lines (not \\) so that any field can be
%  left empty {} — an empty line simply disappears instead of erroring.
\newcommand{\MakeLetterhead}{%
  \noindent
  \begin{minipage}[c]{0.72\textwidth}
    \setlength{\parskip}{3pt}%
    {\fontsize{21}{25}\selectfont\textsc{\color{AccentColor}\SenderName}\par}%
    {\color{AccentColor}\rule{0.95\linewidth}{0.8pt}\par}%
    {\footnotesize\sffamily\color{MutedText}%
      \IfNonEmpty{\Organisation}{\textbf{\Organisation}\par}%
      \AddressBlock{\SenderAddress}%
      \IfNonEmpty{\SenderContact}{\par\SenderContact\par}%
    }%
  \end{minipage}\hfill
  \begin{minipage}[c]{0.24\textwidth} % adjust the emblem size here
    \raggedleft
    \IfFileExists{\LogoPath}%
      {\includegraphics[width=\linewidth]{\LogoPath}}%
      {\color{RuleColor}\rule{\linewidth}{0.8pt}}%  placeholder rule until a logo is dropped in
  \end{minipage}\par
}
