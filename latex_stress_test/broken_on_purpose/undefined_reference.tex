% Not input by main.tex — a second failure shape, kept separate so each fixture
% fails for exactly one reason.
%
% This one *compiles*: an undefined reference and an undefined citation are
% warnings, not errors. That distinction is the whole content of this file. A
% log parser that promotes every complaint to an error makes a working document
% look broken; one that ignores warnings hides the two mistakes users most
% often actually make.
\documentclass[11pt]{article}
\begin{document}

\section{Present}
\label{sec:present}

A reference that resolves: \ref{sec:present}.

% STRESS-WARN-REF: an undefined reference — a warning, never an error.
A reference that does not: \ref{sec:nowhere}.

% STRESS-WARN-CITE: an undefined citation — likewise a warning.
A citation with no bibliography: \cite{nosuchkey}.

\end{document}
